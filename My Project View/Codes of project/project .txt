// Function to draw a filled circle
void drawCircle_4(float cx, float cy, float r) {
    glBegin(GL_TRIANGLE_FAN);
    for (int i = 0; i <= 100; ++i) {
        float theta = 2.0f * 3.1416f * i / 100;
        float x = r * cos(theta);
        float y = r * sin(theta);
        glVertex2f(cx + x, cy + y);
    }
    glEnd();
}

// Function to draw a filled circle with specified color and segments
void drawCircle_4(float cx, float cy, float r, int segments, Color c) {
    glColor3f(c.r, c.g, c.b);
    glBegin(GL_TRIANGLE_FAN);
    glVertex2f(cx, cy);
    for (int i = 0; i <= segments; i++) {
        float angle = 2.0f * M_PI * i / segments;
        glVertex2f(cx + r * cosf(angle), cy + r * sinf(angle));
    }
    glEnd();
}

// New helper function to draw a 5-pointed star
void drawFivePointedStar_4(float cx, float cy, float outerRadius, float innerRadius, Color c) {
    glColor3f(c.r, c.g, c.b);
    glBegin(GL_TRIANGLE_FAN);

    // Center point for the star (optional, makes it easier to fill)
    glVertex2f(cx, cy);

    // Draw 10 vertices for a 5-pointed star (5 outer points, 5 inner points)
    // Adjust angle to make the star upright
    for (int i = 0; i <= 10; ++i) {
        float angle = M_PI / 5.0f * i; // Angle for each segment (2 segments per point)
        float r = (i % 2 == 0) ? outerRadius : innerRadius; // Alternate between outer and inner radius
        glVertex2f(cx + r * cos(angle - M_PI / 2.0f), cy + r * sin(angle - M_PI / 2.0f)); // Rotate to start upright
    }
    glEnd();
}

void initRain_4() {
    for (int i = 0; i < MAX_RAINDROPS; ++i) {
        raindrops[i].x = rand() % 1000;
        raindrops[i].y = rand() % 600;
        raindrops[i].speed = 4.0f + (rand() % 5);
    }
}

void initStars_4() {
    for (int i = 0; i < MAX_STARS; ++i) {
        stars[i].x = rand() % 1000;
        stars[i].y = 200 + rand() % 400; // Stars primarily in the upper sky
        stars[i].size = (rand() % 3) + 1.0f;
    }
}

void drawRain_4() {
    if (!isRaining_4) return;
    glColor3f(0.8f, 0.8f, 1.0f); // Brighter blue for raindrops
    glLineWidth(2);
    glBegin(GL_LINES);
    for (int i = 0; i < MAX_RAINDROPS; ++i) {
        float x = raindrops[i].x;
        float y = raindrops[i].y;
        glVertex2f(x, y);
        glVertex2f(x, y - 15); // Draw a slightly longer line for each drop
    }
    glEnd();
    glLineWidth(1); // Reset line width
}

void drawStars_4() {
    // Only draw stars if it's night and not morning
    if (isDaytime_4 || isMorning_4) return;
    glColor3f(1.0f, 1.0f, 1.0f); // White stars
    glBegin(GL_POINTS);
    for (int i = 0; i < MAX_STARS; ++i) {
        glPointSize(stars[i].size);
        glVertex2f(stars[i].x, stars[i].y);
    }
    glEnd();
    glPointSize(1); // Reset point size
}

void updateRain_4() {
    if (!isRaining_4) return;
    for (int i = 0; i < MAX_RAINDROPS; ++i) {
        raindrops[i].y -= raindrops[i].speed;
        if (raindrops[i].y < 0) { // If raindrop goes off screen, reset it to the top
            raindrops[i].x = rand() % 1000;
            raindrops[i].y = 600;
            raindrops[i].speed = 4.0f + (rand() % 5);
        }
    }
}

void drawRainbow_4() {
    if (!showRainbow_4) return;

    float cx =460, cy=450; // Rainbow center
    float radius = 100;

    float colors[7][3] = {
        {1.0, 0.0, 0.0},  // Red
        {1.0, 0.5, 0.0},  // Orange
        {1.0, 1.0, 0.0},  // Yellow
        {0.0, 1.0, 0.0},  // Green
        {0.0, 0.0, 1.0},  // Blue
        {0.29, 0.0, 0.51},// Indigo
        {0.56, 0.0, 1.0}  // Violet
    };

    for (int j = 0; j < 7; j++) {
        glColor3f(colors[j][0], colors[j][1], colors[j][2]);
        glBegin(GL_LINE_STRIP);
        for (float i = 0; i <= 180; i += 1) {
            float angle = i * 3.14159 / 180;
            float x = cx + cos(angle) * (radius + j * 5);
            float y = cy + sin(angle) * (radius + j * 5);
            glVertex2f(x, y);
        }
        glEnd();
    }
}

void drawBuilding_4(float x, float y, float width, float height) {
    // Building structure
    Color buildingColor;
    Color windowColor;

    if (isMorning_4) {
        buildingColor = buildingColorMorning;
        windowColor = windowColorMorning;
    } else if (isDaytime_4) {
        buildingColor = buildingColorDay;
        windowColor = windowColorDay;

        buildingColor= buildingColorNight;
        windowColor = windowColorNight;
    }

    glColor3f(buildingColor.r, buildingColor.g, buildingColor.b);
    glBegin(GL_QUADS);
    glVertex2f(x, y);
    glVertex2f(x + width, y);
    glVertex2f(x + width, y + height);
    glVertex2f(x, y + height);
    glEnd();

    // Window grid calculation based on building size
    int rows = (int)(height / 40);
    int cols = (int)(width / 50);

    float winWidth = 20.0f;
    float winHeight = 20.0f;
    float xOffset = 15.0f;
    float yOffset = 15.0f;

    // Draw windows
    glColor3f(windowColor.r, windowColor.g, windowColor.b);
    for (int r = 0; r < rows; ++r) {
        for (int c = 0; c < cols; ++c) {
            float wx1 = x + xOffset + c * 40;
            float wy1 = y + yOffset + r * 30;
            float wx2 = wx1 + winWidth;
            float wy2 = wy1 + winHeight;

            glBegin(GL_QUADS);
            glVertex2f(wx1, wy1);
            glVertex2f(wx2, wy1);
            glVertex2f(wx2, wy2);
            glVertex2f(wx1, wy2);
            glEnd();

            // Lights are off during day and morning
            if (!isDaytime_4 && !isMorning_4) {
                drawLight_4(wx1, wy1, wx2, wy2);
            }
        }
    }
}

void update(int value) {
    // Calculate the total width of the train
    float trainLength = 5 * 80.0f + 90.0f; // 5 coaches + engine length (approx)

    trainX_4 += trainSpeed_4;
    if (trainX_4 > 1000 + trainLength) {
        trainX_4 = -trainLength; // Reset to start just off the left side
    }

    cloudPos_4+= 0.3f;
    if (cloudPos_4 > 1100.0f) cloudPos_4 = 0.0f;

    // Adjust sun/moon position logic to account for morning
    if (isMorning_4) {

        if (sunMoonPos_4 < 0.5f) sunMoonPos_4+= 0.0005f;
        else if (sunMoonPos_4> 0.5f) sunMoonPos_4 = 0.5f;
    } else if (isDaytime_4) {

        if (sunMoonPos_4< 1.0f) sunMoonPos_4 += 0.0005f; // Slower movement
        else if (sunMoonPos_4 > 1.0f) sunMoonPos_4 = 1.0f; // Cap at a certain point for "day"
    } else { // Night transition (moon rising/setting)

        if (sunMoonPos_4 < 1.0f) sunMoonPos_4 += 0.0005f; // Slower movement
        else if (sunMoonPos_4 > 1.0f) sunMoonPos_4 = 1.0f;
    }


    // Update boat positions
    boatPos_4+= boatPos_4;
    if (boatPos_4 > 1000.0f + 50.0f) boatPos_4 = -150.0f;

    boat2Pos_4 -= boat2Speed_4; // Boat moving left (opposite direction)
    if (boat2Pos_4 < -150.0f) boat2Pos_4 = 1100.0f;

    boat3Pos_4 += fasterBoatSpeed_4; // Third boat moving right
    if (boat3Pos_4 > 1000.0f + 75.0f) boat3Pos_4 = -300.0f;

    updateRain_4();
    airplanePos_4 += airplaneSpeed_4;
    if (airplanePos_4 > 1000.0f + 120.0f) airplanePos_4 = -200.0f;

    windmillAngle_4+= 1.0f;
    if (windmillAngle_4 >= 360.0f) windmillAngle_4-= 360.0f;

    // --- Lightning Flash Timer ---
    if (isLightning_4) {
        lightningTimer_4--;
        if (lightningTimer_4 <= 0) {
            isLightning_4= false; // Turn off flash when timer expires
        }
    }
    // --- End Lightning Flash Timer ---

    glutPostRedisplay();
    glutTimerFunc(16, update, 0); // ~60 FPS

}

void keyboard(unsigned char key, int x, int y) {
    switch (key) {
        case 'd': case 'D':
            isDaytime_4 = true;
            isMorning_4= false; // Ensure morning is off
            // Reset colors to default day colors explicitly when 'D' is pressed
            skyDay= {0.53f, 0.81f, 0.92f}; // Re-assign values
            groundDay = {0.4f, 0.7f, 0.2f};
            riverColorDay = {0.2f, 0.6f, 0.8f};
            sunMoonPos_4 = 0.0f; // Reset sun position to start of day (right side for setting)
            break;
        case 'n': case 'N':
            isDaytime_4 = false;
            isMorning_4= false; // Ensure morning is off
            sunMoonPos_4= 0.0f; // Reset moon position to start of night (left side for rising)
            break;
        case 'm': case 'M': // New key for morning view
            isMorning_4 = true;
            isDaytime_4 = false; // Ensure it's not also daytime

            // **Darker Morning Sky Colors**
            skyMorning.r = 0.1f;
            skyMorning.g = 0.1f;
            skyMorning.b = 0.2f;

            // Complementary darker ground and river colors for the darker morning sky
            groundMorning.r = 0.25f;
            groundMorning.g = 0.35f;
            groundMorning.b = 0.15f;

            riverColorMorning.r = 0.1f;
            riverColorMorning.g = 0.2f;
            riverColorMorning.b = 0.3f;

            sunMoonPos_4= 0.0f; // Reset sun position to start of morning rise (left side)
        case '+': // Increase speed
    if (trainSpeed_4< 10.0f) trainSpeed_4 += 0.2f;
    //if (boatSpeed_4< 5.0f) boatSpeed_4+= 0.1f;
    if (boat2Speed_4< 4.0f) boat2Speed_4+= 0.1f;
    if (fasterBoatSpeed_4< 6.0f) fasterBoatSpeed_4 += 0.1f;
    break;

case '-': // Decrease speed
    if (trainSpeed_4 > 0.2f) trainSpeed_4 -= 0.2f;
//    if (boatSpeed_4 > 0.1f) boatSpeed_4 -= 0.1f;
    if (boat2Speed_4 > 0.1f) boat2Speed_4 -= 0.1f;
    if (fasterBoatSpeed_4 > 0.1f) fasterBoatSpeed_4 -= 0.1f;
    break;

        case ' ': // Toggle pause/resume
            if (trainSpeed_4 > 0.01f) {
                trainSpeed_4 = 0.0f;
                boat2Speed_4 = 0.0f;
                fasterBoatSpeed_4= 0.0f;
            } else { // Resume to default speeds
                trainSpeed_4 = 2.0f;
                boat2Speed_4 = 0.8f;
                fasterBoatSpeed_4 = 1.2f;
            }
            break;
        case 'l': case 'L': // New: Lightning flash
            // Only allow lightning if it's night or raining
            if (!isDaytime_4 && !isMorning_4 || isRaining_4) {
                isLightning_4= true;
                lightningTimer_4 = LIGHTNING_DURATION_4;
            }
            break;

           // ... other cases ...
        case 'r': case 'R': // Show rainbow
            showRainbow_4 = true;
            glutTimerFunc(3000, rainbowOff_4, 0); // Hide rainbow after 3 seconds
            break;
        // ... other


    }
    glutPostRedisplay(); // Request a redraw after keyboard input
}

void mouse(int button, int state, int x, int y) {
    if (state == GLUT_DOWN) {
        if (button == GLUT_RIGHT_BUTTON) {
            isRaining_4 = true;
        } else if (button == GLUT_LEFT_BUTTON) {
            isRaining_4 = false;
        }
    }
}


void reshape(int w, int h) {
    glViewport(0, 0, w, h);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, 1000, 0, 600); // Set up 2D orthographic projection
    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();
}

int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB); // Double buffering and RGB color
    glutInitWindowSize(1000, 600); // Initial window size
    glutCreateWindow("Animated Scene: Railway, River, City Elements"); // Window title

    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutKeyboardFunc(keyboard);
    glutMouseFunc(mouse);
    glutTimerFunc(0, update, 0);
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    initRain_4();  // Initialize raindrop positions
    initStars_4(); // Initialize star positions

    glutMainLoop(); // Enter the GLUT event processing loop
    return 0;
}
